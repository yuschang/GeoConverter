from ultralytics import YOLO

# Choose your YOLO12 checkpoint: yolov12n.pt, yolov12s.pt, yolov12m.pt, etc.
CHECKPOINT = "yolov12m.pt"

# Create model instance
model = YOLO(CHECKPOINT)

# Train with heavier GPU usage:
results = model.train(
    data=f"{dataset.location}/data.yaml",
    epochs=250,
    imgsz=640,

    # —— GPU / performance params ——
    # use both GPUs 0 and 1 (or list your GPU IDs)
    device='0',

    # increase batch size to fill more GPU memory (tune this to your GPU VRAM)
    batch= 32,

    # number of DataLoader workers for faster I/O
    workers=8,

    # cache images in RAM for ultra-fast sampling
    cache=True,

    # mixed-precision training (speeds up FLOPs and reduces memory)
    amp=True,

    name="roboflow_yolov12_run"
)
